% PLACEHOLDERS (replaced by career-ops at generation time):
%   {{NAME}}              — Full name
%   {{EMAIL}}             — Email address
%   {{LINKEDIN_URL}}      — LinkedIn profile URL
%   {{LINKEDIN_DISPLAY}}  — LinkedIn display text
%   {{GITHUB_URL}}        — GitHub profile URL
%   {{GITHUB_DISPLAY}}    — GitHub display text
%   {{STACKOVERFLOW_URL}} — StackOverflow profile URL
%   {{STACKOVERFLOW_DISPLAY}} — StackOverflow display text
%   {{CTFTIME_URL}}       — CTFtime team URL
%   {{CTFTIME_DISPLAY}}   — CTFtime display text
%   {{PHONE}}             — Phone number
%   {{SUMMARY}}           — Summary paragraph (no \small{} wrapper needed)
%   {{EXPERIENCE}}        — All experience blocks (resumeSubheading + resumeItemListStart/End)
%   {{EDUCATION}}         — Education block (resumeSubheading)
%   {{SKILLS}}            — Skills items (\item + \textbf{Category}{: items})
%   {{ACHIEVEMENTS}}      — Achievement items (resumeItem blocks)
%------------------------

% Custom commands
\newcommand{\resumeItem}[2]{
  \item\small{
    \textbf{#1}{: #2 \vspace{-2pt}}
  }
}

\newcommand{\resumeItemNoBullet}[2]{
  \item[]\small{
    \hspace{-9pt}\textbf{#1}{: #2 \vspace{-6pt}}
  }
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item[]
  \begin{tabular*}{0.98\textwidth}{l@{\extracolsep{\fill}}r}
      \hspace{-10pt}\textbf{#1} & #2 \\
      \hspace{-10pt}\textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-5pt}
}

\newcommand{\resumeSubItem}[2]{\resumeItem{#1}{#2}\vspace{-4pt}}

\renewcommand{\labelitemii}{$\circ$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=*]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}

% custom commands
\newcommand{\shorterSection}[1]{\vspace{-4pt}\section{#1}}

\begin{document}
    \begin{center}
        % NAME
        {\Huge\scshape {{NAME}}}
        % SUBHEADING
        \\ {{CONTACT_LINE}}\\
        \small
        % EMAIL
        % EMAIL_URL already carries its scheme: sanitizeUrl() prefixes bare
        % addresses with mailto:, so the template must not add a second one.
        \href{{{EMAIL_URL}}}{\raisebox{-0.2\height}\faEnvelope\  \underline{{{EMAIL_DISPLAY}}}} ~
        % LINKEDIN
        \href{{{LINKEDIN_URL}}}{\raisebox{-0.2\height}\faLinkedin\ \underline{{{LINKEDIN_DISPLAY}}}}  ~
        % GITHUB
        \href{{{GITHUB_URL}}}{\raisebox{-0.2\height}\faGithub\ \underline{{{GITHUB_DISPLAY}}}}
    \end{center}

%----------HEADING-----------------
\begin{center}
  \small \textbf{\huge {{NAME}}}  \\[5pt]
  \href{mailto:{{EMAIL}}}
  {\faicon{envelope} \underline{{{EMAIL}}}}  $\vert$
  \faicon{linkedin} \href{{{LINKEDIN_URL}}}{\underline{{{LINKEDIN_DISPLAY}}}} $\vert$
   \faicon{github} \href{{{GITHUB_URL}}}{\underline{{{GITHUB_DISPLAY}}}} $\vert$
  \faicon{stack-overflow} \href{{{STACKOVERFLOW_URL}}}{\underline{{{STACKOVERFLOW_DISPLAY}}}} $\vert$
  \faicon{flag} \href{{{CTFTIME_URL}}}{\underline{{{CTFTIME_DISPLAY}}}} $\vert$
  {{PHONE}}\\
\end{center}
